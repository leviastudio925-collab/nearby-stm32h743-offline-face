cmake_minimum_required(VERSION 3.16)
project(nearby_offline_face C)
set(CMAKE_C_STANDARD 11)
add_library(nearby_face src/nearby_face.c)
target_include_directories(nearby_face PUBLIC include)
add_executable(test_face tests/test_face.c)
target_link_libraries(test_face PRIVATE nearby_face)
if(MSVC)
  target_compile_options(test_face PRIVATE /UNDEBUG)
else()
  target_compile_options(test_face PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME offline_face_core COMMAND test_face)
