   The MEMORY region RAM_D1 must refer to AXI SRAM at 0x24000000 (512 KiB).
   Do NOT place these sections in DTCMRAM at 0x20000000.
   Check the final .map file and memory consumption after linking. */

{

  *(.camera_dma)
  . = ALIGN(32);
} >RAM_D1

{

  *(.face_work)
  . = ALIGN(32);
} >RAM_D1
